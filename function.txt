def predict_repo(repo_string, headers, threshold, rf_model, tfidf_readme, tfidf_topics, scaler):
    """
    Process a single repo and return a dict with prediction data.
    On error, returns a dict with 'error' key set.
    """
    try:
        features = fetch_repo_features(repo_string, headers=headers)
    except RepoFetchError as e:
        return {"repo": repo_string, "error": str(e)}
    except Exception as e:
        return {"repo": repo_string, "error": f"Unexpected error: {e}"}

    topics = features.get("topics", [])
    repo_age_days = features.get("repo_age_days", 0)

    X_dense = featurize(features, tfidf_readme, tfidf_topics, scaler)

    probability = float(rf_model.predict_proba(X_dense)[0][1])
    prediction = 1 if probability >= threshold else 0

    # Log to audit trail and compute caveats
    exceptions = check_exceptions(features)
    low_confidence = 0.4 <= probability <= 0.6
    warning_messages = exceptions.copy()
    if low_confidence:
        warning_messages.append("Low confidence prediction (probability near 0.5).")
    # Note: we will log after we have all data, but we need to compute combined score etc. first?
    # We'll compute heuristic and combined scores now.
    try:
        from scoring_engine import RepoScorer
        scorer = RepoScorer()
        heuristic_result = scorer.calculate_score(features)
        ml_score_pct = probability * 100
        heuristic_score = heuristic_result['total_score']
        combined_score = (ml_score_pct + heuristic_score) / 2
        divergence = abs(ml_score_pct - heuristic_score)
    except Exception:
        heuristic_result = None
        ml_score_pct = probability * 100
        heuristic_score = None
        combined_score = None
        divergence = None

    # Prepare data for logging
    timestamp = datetime.now().isoformat()
    repo_id = features.get("full_name", "")
    logged_features = {
        "full_name": features.get("full_name", ""),
        "html_url": features.get("html_url", ""),
        "stars": features.get("stars", 0),
        "forks": features.get("forks", 0),
        "open_issues": features.get("open_issues", 0),
        "readme_size": features.get("readme_size", 0),
        "repo_age_days": features.get("repo_age_days", 0),
        "last_commit_days": features.get("last_commit_days", 0),
        "has_readme": features.get("has_readme", 0),
        "topics_count": len(features.get("topics", [])),
        "probability": f"{probability:.6f}",
        "prediction": prediction,
        "threshold": f"{threshold:.2f}",
        "timestamp": timestamp
    }
    # Log to CSV and JSON Lines (we'll reuse log_audit_trail but we need to avoid duplicate logging? 
    # We'll just call log_audit_trail here.
    log_audit_trail(features, probability, prediction, threshold, caveats=warning_messages)

    # Build result dict
    result = {
        "repo": repo_string,
        "repo_full_name": features.get("full_name", ""),
        "repo_url": features.get("html_url", ""),
        "stars": features.get("stars", 0),
        "forks": features.get("forks", 0),
        "open_issues": features.get("open_issues", 0),
        "repo_age_days": features.get("repo_age_days", 0),
        "topics_count": len(features.get("topics", [])),
        "probability": probability,
        "prediction": prediction,
        "ml_score_pct": ml_score_pct,
        "heuristic_score": heuristic_score,
        "combined_score": combined_score,
        "divergence": divergence,
        "exceptions": exceptions,
        "low_confidence": low_confidence,
        "warning_messages": warning_messages,
        "error": None
    }
    return result
